package dailyassignments;

class Student
{
	int rollNo;
	String name;
	static String collageName = "ABC Engineering Collage";
	static int studentCount = 0;
		
static
{
	System.out.println("student class initialized.");
}

public Student(int rollNo, String name)
{
	this.rollNo = rollNo;
	this.name = name;
	studentCount++;

}

public static void changeCollageName(String newCollageName)
{
	collageName = newCollageName;
	System.out.println("collage name updated successfully");
}

public void displayDetails()
{
	System.out.println("Roll No: "+rollNo);
	System.out.println("Name: "+name);
	System.out.println("Collage Name: "+collageName);
	System.out.println();
}

public static void displayStudentCount()

{
	System.out.println();
	System.out.println("Total Student: "+ studentCount);
}

}


public class StudentMgmtUsingStatic {

	public static void main(String[] args) {
		// TODO Auto-generated method stub
		System.out.println();
		Student s1 = new Student(101, "Rahul");
		Student s2 = new Student(102, "Priya");
		Student s3 = new Student(103, "Amit");
		
		s1.displayDetails();
		s2.displayDetails();
		s3.displayDetails();
		System.out.println();
		
		Student.displayStudentCount();
		
		
		s1.displayDetails();
		s2.displayDetails();
		s3.displayDetails();
		
		
				
	}

}
